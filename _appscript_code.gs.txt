// ---------- OAUTH ----------
function getAccessToken_(){
  var p = props_();
  var url = accountsBase_() + '/oauth/v2/token';
  var payload = {
    refresh_token: p.getProperty('REFRESH_TOKEN'),
    client_id: p.getProperty('CLIENT_ID'),
    client_secret: p.getProperty('CLIENT_SECRET'),
    grant_type: 'refresh_token'
  };
  var res = UrlFetchApp.fetch(url, {method:'post', payload:payload, muteHttpExceptions:true});
  var j = JSON.parse(res.getContentText());
  if(!j.access_token) throw new Error('Token refresh failed: ' + res.getContentText());
  return j.access_token;
}

/** ONE-TIME setup: put CLIENT_ID, CLIENT_SECRET, GRANT_CODE, ZOHO_DC in Script Properties, then run this once. */
function exchangeGrantForRefresh(){
  var p = props_();
  var url = accountsBase_() + '/oauth/v2/token';
  var payload = {
    grant_type:'authorization_code',
    client_id:p.getProperty('CLIENT_ID'),
    client_secret:p.getProperty('CLIENT_SECRET'),
    code:p.getProperty('GRANT_CODE')
  };
  var res = UrlFetchApp.fetch(url, {method:'post', payload:payload, muteHttpExceptions:true});
  var j = JSON.parse(res.getContentText());
  if(j.refresh_token){
    p.setProperty('REFRESH_TOKEN', j.refresh_token);
    p.deleteProperty('GRANT_CODE');
    return 'OK: refresh token stored. You can delete GRANT_CODE.';
  }
  throw new Error('Exchange failed: ' + res.getContentText());
}

// ---------- CRM ----------
function coql_(token, query){
  var res = UrlFetchApp.fetch(crmBase_() + '/coql', {
    method:'post',
    contentType:'application/json',
    headers:{Authorization:'Zoho-oauthtoken ' + token},
    payload: JSON.stringify({select_query: query}),
    muteHttpExceptions:true
  });
  if(res.getResponseCode()===204) return [];
  var j = JSON.parse(res.getContentText());
  return (j && j.data) ? j.data : [];
}

function fetchOpenCases_(token){
  return coql_(token, "select id, Subject, Status, Account_Name, Deal_Name from Cases where Status != 'Closed' limit 200");
}
function fetchMilestoneTasks_(token){
  return coql_(token, "select id, Subject, Status, Due_Date, What_Id from Tasks where Subject like '[M%' limit 200");
}

// ---------- DESK ----------
function deskRequests_(token){
  var reqs = [];
  Object.keys(INS).forEach(function(k){
    var d = INS[k].desk; if(!d) return;
    [['it',DEPT.it],['ops1',DEPT.ops1],['ops2',DEPT.ops2]].forEach(function(pair){
      var url = deskBase_() + '/tickets/search?departmentId=' + pair[1]
              + '&status=' + encodeURIComponent('Open,In Progress')
              + '&accountName=' + encodeURIComponent(d) + '&limit=99&from=0';
      reqs.push({key:k, bucket:pair[0], request:{url:url, method:'get',
        headers:{Authorization:'Zoho-oauthtoken ' + token, orgId: deskOrg_()}, muteHttpExceptions:true}});
    });
  });
  return reqs;
}
function fetchDesk_(token){
  var reqs = deskRequests_(token);
  var responses = UrlFetchApp.fetchAll(reqs.map(function(r){return r.request;}));
  var out = {}; // key -> {it,ops,client}
  Object.keys(INS).forEach(function(k){ if(INS[k].desk) out[k] = {it:0, ops:0, client:0}; });
  reqs.forEach(function(r, i){
    var body = responses[i].getContentText();
    var n = 0;
    try { var j = JSON.parse(body); n = (typeof j.count === 'number') ? j.count : ((j.data||[]).length); } catch(e){ n = 0; }
    if(r.bucket==='it') out[r.key].it += n; else out[r.key].ops += n;
  });
  return out;
}

// ---------- ASSEMBLE ----------
function getPayload(){
  var token = getAccessToken_();
  var cases = fetchOpenCases_(token);
  var tasks = fetchMilestoneTasks_(token);
  var desk = {};
  try { desk = fetchDesk_(token); } catch(e){ desk = {__error:String(e)}; }

  // dealId -> code, crmAccId -> insKey
  var dealToCode = {}; Object.keys(MS).forEach(function(c){ dealToCode[MS[c].dealId] = c; });
  var accToIns = {}; Object.keys(INS).forEach(function(k){ if(INS[k].crmId) accToIns[INS[k].crmId] = k; });
  // primary milestone per insurer (first MS whose ins==k)
  var primaryMs = {}; Object.keys(MS).forEach(function(c){ var k=MS[c].ins; if(!primaryMs[k]) primaryMs[k]=c; });

  // Group open cases by milestone code
  var casesByMs = {};
  cases.forEach(function(c){
    var code = null;
    if(c.Deal_Name && c.Deal_Name.id && dealToCode[c.Deal_Name.id]) code = dealToCode[c.Deal_Name.id];
    if(!code && c.Account_Name && c.Account_Name.id && accToIns[c.Account_Name.id]) code = primaryMs[accToIns[c.Account_Name.id]];
    if(!code) code = '_unlinked';
    (casesByMs[code] = casesByMs[code] || []).push({id:c.id, subject:c.Subject, status:c.Status});
  });

  // Group OPEN tasks by milestone via What_Id (deal) or subject prefix
  var tasksByMs = {};
  tasks.forEach(function(t){
    var st = (t.Status||'').toLowerCase();
    if(st==='completed' || st==='closed') return; // only open tasks
    var code = null;
    if(t.What_Id && t.What_Id.id && dealToCode[t.What_Id.id]) code = dealToCode[t.What_Id.id];
    if(!code){ var m = /^\[(M\d+)/.exec(t.Subject||''); if(m) code = m[1]; }
    if(!code) return;
    (tasksByMs[code] = tasksByMs[code] || []).push({id:t.id, subject:t.Subject, due:t.Due_Date||'', status:t.Status});
  });

  return JSON.stringify({
    updated: Utilities.formatDate(new Date(), 'Asia/Kolkata', "EEE dd MMM yyyy, HH:mm 'IST'"),
    INS:INS, MS:MS, desk:desk, casesByMs:casesByMs, tasksByMs:tasksByMs,
    crmOrg:'org55302521'
  });
}

function doGet(){
  var t = HtmlService.createTemplateFromFile('Index');
  t.payload = getPayload().replace(/</g, '\\u003c'); // safe to inline as a JS object literal
  return t.evaluate().setTitle('CT Insurer Command Page')
    .addMetaTag('viewport','width=device-width, initial-scale=1.0');
}

// Called by the Refresh button (client) to re-pull live and get fresh data.
function refresh(){ return getPayload(); }
